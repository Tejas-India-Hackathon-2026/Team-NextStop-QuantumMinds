const API = "http://127.0.0.1:8000";

let recipients = [];

let payer = null;


/* ============================================================
   LOAD PAYER
============================================================ */

async function loadProfile() {

    try {

        const response =
            await fetch(
                API + "/profile"
            );

        if (!response.ok) {

            throw new Error(
                "Could not load payer."
            );
        }

        payer =
            await response.json();

        document.getElementById(
            "payerName"
        ).textContent =
            payer.name;

        document.getElementById(
            "avatar"
        ).textContent =
            payer.name
                .charAt(0)
                .toUpperCase();

    }

    catch (error) {

        showError(
            "Could not load payer from database: "
            + error.message
        );

    }

}


/* ============================================================
   LOAD RECIPIENTS
============================================================ */

async function loadRecipients() {

    try {

        const response =
            await fetch(
                API + "/recipients"
            );

        if (!response.ok) {

            throw new Error(
                "Could not load recipients."
            );
        }

        recipients =
            await response.json();

        const select =
            document.getElementById(
                "recipient"
            );

        select.innerHTML =
            '<option value="">Select recipient</option>';

        recipients.forEach(
            recipient => {

                const option =
                    document.createElement(
                        "option"
                    );

                option.value =
                    recipient.id;

                option.textContent =
                    recipient.name
                    +
                    (
                        recipient.upi_id
                            ? " • "
                            + recipient.upi_id
                            : ""
                    );

                select.appendChild(
                    option
                );

            }
        );

    }

    catch (error) {

        showError(
            "Could not load recipients from database: "
            + error.message
        );

    }

}

/* ============================================================
   DEVICE ID
============================================================ */

function getDeviceId() {

    let deviceId =
        localStorage.getItem(
            "secureflow_device_id"
        );

    if (!deviceId) {

        deviceId =
            "device-" +
            Math.random()
                .toString(36)
                .substring(2)
            +
            Date.now()
                .toString(36);

        localStorage.setItem(
            "secureflow_device_id",
            deviceId
        );

    }

    return deviceId;
}


/* ============================================================
   GET LOCATION AUTOMATICALLY
============================================================ */

function getLocation() {

    return new Promise(
        resolve => {

            if (
                !navigator.geolocation
            ) {

                resolve({
                    latitude: null,
                    longitude: null
                });

                return;

            }

            navigator.geolocation.getCurrentPosition(

                position => {

                    resolve({

                        latitude:
                            position.coords.latitude,

                        longitude:
                            position.coords.longitude

                    });

                },

                () => {

                    resolve({

                        latitude: null,
                        longitude: null

                    });

                },

                {
                    enableHighAccuracy: false,
                    timeout: 4000,
                    maximumAge: 300000
                }

            );

        }
    );

}


/* ============================================================
   ANALYZE PAYMENT
============================================================ */

async function analyzePayment() {

    clearError();

    const amount =
        Number(
            document.getElementById(
                "amount"
            ).value
        );

    const recipient =
        document.getElementById(
            "recipient"
        ).value;

    if (!recipient) {

        showError(
            "Please select a recipient."
        );

        return;

    }

    if (!amount || amount <= 0) {

        showError(
            "Please enter a valid amount."
        );

        return;

    }


    const button =
        document.getElementById(
            "payButton"
        );

    button.disabled = true;

    button.textContent =
        "🔍 Analysing behaviour...";


    const location =
        await getLocation();


    try {

        const response =
            await fetch(
                API + "/transaction",
                {

                    method: "POST",

                    headers: {
                        "Content-Type":
                            "application/json"
                    },

                    body:
                        JSON.stringify({

                            amount:
                                amount,

                            recipient_id:
                                recipient,

                            device_id:
                                getDeviceId(),

                            timezone:
                                Intl
                                    .DateTimeFormat()
                                    .resolvedOptions()
                                    .timeZone,

                            latitude:
                                location.latitude,

                            longitude:
                                location.longitude

                        })

                }
            );


        const data =
            await response.json();


        if (!response.ok) {

            throw new Error(
                data.detail
                ||
                "Transaction analysis failed."
            );

        }


        showResult(data);

    }

    catch (error) {

        showError(
            error.message
        );

    }

    finally {

        button.disabled =
            false;

        button.textContent =
            "🔒 Pay Securely";

    }

}


/* ============================================================
   SHOW RESULT
============================================================ */

function showResult(data) {

    const result =
        document.getElementById(
            "result"
        );

    result.classList.add(
        "show"
    );


    document.getElementById(
        "score"
    ).textContent =
        data.risk_score;


    const decision =
        document.getElementById(
            "decision"
        );


    if (
        data.decision
        ===
        "ALLOW"
    ) {

        decision.textContent =
            "✓ PAYMENT APPROVED";

        decision.style.color =
            "#159447";

    }

    else if (
        data.decision
        ===
        "ALERT"
    ) {

        decision.textContent =
            "⚠ PAYMENT HELD";

        decision.style.color =
            "#c26a00";

    }

    else {

        decision.textContent =
            "✕ PAYMENT BLOCKED";

        decision.style.color =
            "#c62828";

    }


    document.getElementById(
        "status"
    ).textContent =
        data.status;


    /* SIGNALS */

    const signals =
        document.getElementById(
            "signals"
        );

    signals.innerHTML = "";


    if (
        data.signals
        &&
        data.signals.length > 0
    ) {

        data.signals.forEach(
            signal => {

                const row =
                    document.createElement(
                        "div"
                    );

                row.className =
                    "signal";

                row.innerHTML =

                    `<span>
                        ${escapeHtml(signal.name)}
                    </span>

                    <span class="signal-score">
                        +${signal.score}
                    </span>`;

                signals.appendChild(
                    row
                );

            }
        );

    }


    /* DYNAMIC QUESTIONS
       ONLY IF RISK > 50
    */

    const questions =
        document.getElementById(
            "questions"
        );

    const questionList =
        document.getElementById(
            "questionList"
        );

    questionList.innerHTML = "";


    if (
        data.risk_score > 50
        &&
        data.dynamic_questions
        &&
        data.dynamic_questions.length > 0
    ) {

        questions.style.display =
            "block";


        data.dynamic_questions.forEach(
            (question, index) => {

                const div =
                    document.createElement(
                        "div"
                    );

                div.className =
                    "question";

                div.textContent =
                    (index + 1)
                    +
                    ". "
                    +
                    question;

                questionList.appendChild(
                    div
                );

            }
        );

    }

    else {

        questions.style.display =
            "none";

    }


    result.scrollIntoView({
        behavior: "smooth",
        block: "start"
    });

}


/* ============================================================
   ERROR
============================================================ */

function showError(message) {

    document.getElementById(
        "error"
    ).innerHTML =
        `<div class="error">
            ${escapeHtml(message)}
        </div>`;

}


function clearError() {

    document.getElementById(
        "error"
    ).innerHTML = "";

}


function escapeHtml(text) {

    const div =
        document.createElement(
            "div"
        );

    div.textContent =
        text;

    return div.innerHTML;

}
