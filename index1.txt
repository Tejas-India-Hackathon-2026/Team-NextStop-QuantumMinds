<script>

function addTransaction(tx) {

    const row = document.createElement("tr");

    row.innerHTML = `
        <td>${tx.transaction_id}</td>
        <td>${tx.user_id}</td>
        <td>₹${tx.amount}</td>
        <td>${tx.risk_score}</td>
        <td class="${tx.action.toLowerCase()}">
            ${tx.action}
        </td>
    `;

    document
        .getElementById("transactions")
        .prepend(row);
}

setInterval(() => {

    const actions = ["ALLOW", "ALERT", "BLOCK"];

    const action =
        actions[Math.floor(Math.random() * actions.length)];

    addTransaction({
        transaction_id:
            "TXN" + Math.floor(Math.random() * 100000),

        user_id: "USER001",

        amount:
            Math.floor(Math.random() * 10000) + 100,

        risk_score:
            Math.floor(Math.random() * 100),

        action: action
    });

}, 3000);
