        function previewRisk() {

            if (
                signal("frequency_anomaly")

            if (
                signal("unusual_location")
            ) score += 15;

            if (
                signal("sudden_location_change")
            ) score += 6;

            if (
                signal("unknown_beneficiary")
            ) score += 12;

            if (
                signal("no_previous_transaction")
            ) score += 13;

            if (
                signal("unusual_beneficiary_amount")
            ) score += 10;

            score =
                Math.min(
                    100,
                    score
                );

            animateRisk(score);

        }


        // ==================================================
        // RISK ANIMATION
        // ==================================================

        function animateRisk(target) {

            const start =
                Number(
                    riskNumber.textContent
                );

            const duration =
                550;

            const startTime =
                performance.now();

            function update(now) {

                const progress =
                    Math.min(
                        1,
                        (now - startTime)
                        /
                        duration
                    );

                const value =
                    Math.round(
                        start
                        +
                        (
                            target - start
                        )
                        *
                        progress
                    );

                riskNumber.textContent =
                    value;

                const degrees =
                    value
                    *
                    3.6;

                riskCircle.style.background =
                    `
                    conic-gradient(
                        ${
                            value >= 50
                            ? "#ffad33"
                            : "#5b5cf0"
                        }
                        ${degrees}deg,
                        #292c46
                        ${degrees}deg
                    )
                    `;

                if (
                    progress < 1
                ) {

                    requestAnimationFrame(
                        update
                    );

                }

            }

            requestAnimationFrame(
                update
            );


            if (target < 50) {

                decision.className =
                    "decision safe";

                decision.textContent =
                    target === 0
                    ? "Ready for secure analysis"
                    : "Low risk • Payment can proceed";

            }

            else {

                decision.className =
                    "decision warning";

                decision.textContent =
                    "Verification required • Payment will be held";
            }
        }

        async function startPayment() {

            const selectedAmount =
                Number(
                    amount.value
                );

            if (
                !selectedAmount ||
                selectedAmount <= 0
            ) {

                toast(
                    "Enter a valid payment amount"
                );

                amount.focus();

                return;
            }


            if (
                !payer.value
            ) {

                toast(
                    "Select a payer"
                );

                return;
            }


            if (
                !recipient.value
            ) {

                toast(
                    "Select a recipient"
                );

                return;
            }


            payBtn.disabled = true;

            showProcessing();

            previewRisk();


            try {

                const response =
                    await fetch(
                        `${API}/transaction`,
                        {

                            method: "POST",

                            headers: {
                                "Content-Type":
                                    "application/json"
                            },

                            body:
                                JSON.stringify({

                                    user_id:
                                        payer.value,

                                    recipient_upi_id:
                                        recipient.value,

                                    amount:
                                        selectedAmount,

                                    amount_deviation:
                                        signal(
                                            "amount_deviation"
                                        ),

                                    time_anomaly:
                                        signal(
                                            "time_anomaly"
                                        ),

                                    frequency_anomaly:
                                        signal(
                                            "frequency_anomaly"
                                        ),

                                    new_device:
                                        signal(
                                            "new_device"
                                        ),

                                    unusual_location:
                                        signal(
                                            "unusual_location"
                                        ),

                                    sudden_location_change:
                                        signal(
                                            "sudden_location_change"
                                        ),

                                    unknown_beneficiary:
                                        signal(
                                            "unknown_beneficiary"
                                        ),

                                    no_previous_transaction:
                                        signal(
                                            "no_previous_transaction"
                                        ),

                                    unusual_beneficiary_amount:
                                        signal(
                                            "unusual_beneficiary_amount"
                                        )

                                })
                        }
                    );


                const data =
                    await response.json();


                if (!response.ok) {

                    throw new Error(
                        data.detail
                        ||
                        "Transaction failed"
                    );

                }


                currentTransactionId =
                    data.transaction_id;


                animateRisk(
                    data.risk_score
                );


                // ==========================================
                // LOW RISK
                // ==========================================

                if (
                    data.risk_score < 50
                ) {

                    setTimeout(

                            showResult(

                                data

                // HIGH RISK

                    false;

        // ==================================================
        // SUBMIT ANSWER
        // ==================================================

        async function submitAnswer() {

            const answerElement =
                document.getElementById(
                    "answer"
                );


            if (!answerElement) {
                return;
            }


            const answer =
                answerElement.value.trim();


            if (!answer) {

                toast(
                    "Please answer the security question"
                );

                answerElement.focus();

                return;
            }


            const button =
                document.querySelector(
                    ".verify-btn"
                );


            if (button) {

                button.disabled =
                    true;

                button.textContent =
                    "Verifying...";
            }


            try {

                const response =
                    await fetch(
                        `${API}/verify`,
                        {

                            method: "POST",

                            headers: {
                                "Content-Type":
                                    "application/json"
                            },

                            body:
                                JSON.stringify({

                                    transaction_id:
                                        currentTransactionId,

                                    answer:
                                        answer

                                })

                        }
                    );


                const data =
                    await response.json();


                if (!response.ok) {

                    throw new Error(
                        data.detail
                        ||
                        "Verification failed"
                    );

                }


                if (
                    data.verified
                ) {

                    showResult(
                        true,
                        data.message,
                        data
                    );

                }

                else {

                    showResult(
                        false,
                        data.message,
                        data
                    );

                }

            }

            catch(error) {

                toast(
                    error.message
                );

                if (button) {

                    button.disabled =
                        false;

                    button.textContent =
                        "Verify & Continue";
                }

            }

        }
