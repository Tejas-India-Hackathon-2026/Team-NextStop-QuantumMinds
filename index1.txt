<script>


/* LOAD DATA */

const recipient =
localStorage.getItem("recipient");

const upi =
localStorage.getItem("upi");

const amount =
localStorage.getItem("amount");


if(recipient){

document.getElementById("recipient")
.innerText = recipient;

}

if(upi){

document.getElementById("upi")
.innerText = upi;

}

if(amount){

document.getElementById("amount")
.innerText = "₹" + amount;

document.getElementById("questionAmount")
.innerText = "₹" + amount;

}


/* VERIFICATION */

document
.getElementById("verificationForm")
.addEventListener(
"submit",
function(event){

event.preventDefault();


const q1 =
document.querySelector(
'input[name="q1"]:checked'
);

const q2 =
document.querySelector(
'input[name="q2"]:checked'
);

const q3 =
document.querySelector(
'input[name="q3"]:checked'
);


const result =
document.getElementById("result");


/* CHECK ALL ANSWERS */

if(!q1 || !q2 || !q3){

result.innerHTML =
"Please answer all three security questions.";

result.className =
"result failed";

result.style.display =
"block";

return;

}


/*
DEMO DECISION

For this prototype, the transaction
is approved when all answers are YES.
*/

if(
q1.value === "yes" &&
q2.value === "yes" &&
q3.value === "yes"
){

result.innerHTML =
"✓ Verification Successful!<br>" +
"Your transaction has been verified. Redirecting...";

result.className =
"result success";

result.style.display =
"block";


setTimeout(function(){

window.location.href =
"dashboard.html";

},1800);

}

else{

result.innerHTML =
"⚠ Verification Failed.<br>" +
"The transaction has been stopped for your protection.";

result.className =
"result failed";

result.style.display =
"block";


setTimeout(function(){

window.location.href =
"alert.html";

},2200);

}

});


</script>
